# Shared phone-input boundary table -- Dart <-> Kotlin.
#
# BOTH sides read THIS file and must reach the same verdict on every line:
#   test/core/services/contact_number_input_parity_test.dart              (Dart)
#   android/app/src/test/kotlin/com/poyrazoncel/korubeni/emergency/
#       ContactNumberInputParityTest.kt                                    (Kotlin)
#
# The file is printable ASCII only. Every invisible or non-ASCII character is
# written as \uXXXX (UTF-16 code unit); a meaningful space at a field edge is
# written as \u0020 because fields are trimmed.
#
# Vector line, five fields separated by " | ":
#   id | input | sanitized | strict | boundary
#     sanitized = sanitizeContactInput(input)
#     strict    = normalizedCallableOrNull(input)      dispatch validator, UNCHANGED
#     boundary  = normalizedContactInputOrNull(input)  picker + manual entry boundary
#     REJECT    = null
#
# Rule line (starts with @): the COMPLETE definition of the sanitizer. Both
# sides sweep every code point U+0000..U+FFFF against these rules.
#   @delete XXXX[-YYYY]  the code point is removed
#   @space  XXXX[-YYYY]  the code point becomes an ASCII space (U+0020)
#   @fold   XXXX-YYYY    the code point becomes ASCII (cp - 0xFEE0); a fold
#                        that yields '/' becomes a space as well
#   Any code point not listed is kept as-is, so the strict validator still
#   decides on letters, '*', '#', ',', ';', 'p', 'w', URIs and extensions.

# Invisible formatting: soft hyphen, Arabic letter mark, Mongolian vowel
# separator, zero-width characters and LRM/RLM, embeddings and overrides,
# word joiner and invisible operators, isolates and deprecated format
# characters, BOM.
@delete 00AD
@delete 061C
@delete 180E
@delete 200B-200F
@delete 202A-202E
@delete 2060-2064
@delete 2066-206F
@delete FEFF

# Tab, '/', and every Unicode space separator (general category Zs).
@space 0009
@space 002F
@space 00A0
@space 1680
@space 2000-200A
@space 202F
@space 205F
@space 3000

# footnote digit must never join the number.
